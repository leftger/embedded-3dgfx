186    /// Snapshot the engine's raster configuration for a frame of the given size.
187    ///
188    /// This is the single place where engine-level effects (fog / dither / tint /
189    /// palette / stipple / texture mapping / sky / camera direction) are lowered
190    /// into the [`RasterState`](crate::pipeline::rasterize::draw::state::RasterState)
191    /// that both the rasterizers *and* the execute driver consume, so there is one
192    /// representation of the pass rather than one per entry point. Use it to drive
193    /// [`draw_zbuffered_with_state`](crate::pipeline::rasterize::draw::zbuffered::draw_zbuffered_with_state)
194    /// or a custom rasterizer with exactly the effects the engine would apply.
195    #[must_use]
196    pub fn raster_state(
197        &self,
198        width: usize,
199        height: usize,
200    ) -> crate::pipeline::rasterize::draw::state::RasterState<'_> {
201        crate::pipeline::rasterize::draw::state::RasterState::new(width, height)
202            .with_fog(self.fog.as_ref())
203            .with_dither(self.dither.as_ref())
204            .with_screen_tint(self.screen_tint)
205            .with_stipple_mode(self.stipple_mode)
206            .with_palette_mode(self.palette_mode)
207            .with_texture_mapping(self.texture_mapping)
208            .with_sky(self.sky)
209            .with_camera_dir({
210                let d = self.camera.get_direction();
211                [d.x, d.y, d.z]
212            })
213    }

286    pub(crate) fn resolve_render_mode(&self, mode: &RenderMode) -> RenderMode {
287        #[cfg(not(feature = "lighting"))]
288        {
289            let _ = self;
290            mode.clone()
291        }
292        #[cfg(feature = "lighting")]
293        {
294            use crate::config::{MaterialProfile, QualityTier};
295            match self.quality_tier {
296                QualityTier::Fastest => match mode {
297                    #[cfg(feature = "lighting")]
298                    RenderMode::BlinnPhong { .. }
299                    | RenderMode::GouraudLightDir(_)
300                    | RenderMode::Toon(_, _)
301                    | RenderMode::SolidLightDir(_) => RenderMode::Solid,
302                    _ => mode.clone(),
303                },
304                QualityTier::Balanced => match (self.material_profile, mode) {
305                    (MaterialProfile::Unlit, RenderMode::BlinnPhong { .. })
306                    | (MaterialProfile::Unlit, RenderMode::GouraudLightDir(_))
307                    | (MaterialProfile::Unlit, RenderMode::Toon(_, _))
308                    | (MaterialProfile::Unlit, RenderMode::SolidLightDir(_)) => RenderMode::Solid,
309                    (MaterialProfile::Lambert, RenderMode::BlinnPhong { light_dir, .. }) => {
310                        RenderMode::SolidLightDir(*light_dir)
311                    }
312                    _ => mode.clone(),
313                },
314                QualityTier::Quality => match (self.material_profile, mode) {
315                    (MaterialProfile::Unlit, RenderMode::BlinnPhong { .. })
316                    | (MaterialProfile::Unlit, RenderMode::GouraudLightDir(_))
317                    | (MaterialProfile::Unlit, RenderMode::Toon(_, _))
318                    | (MaterialProfile::Unlit, RenderMode::SolidLightDir(_)) => RenderMode::Solid,
319                    (MaterialProfile::Lambert, RenderMode::BlinnPhong { light_dir, .. }) => {
320                        RenderMode::SolidLightDir(*light_dir)
321                    }
322                    _ => mode.clone(),
323                },
324            }
325        }
326    }

342    /// Project a 3D world position into 2D screen coordinates using the camera's view-projection matrix.
343    pub fn project_point(
344        &self,
345        point: Point3<f32>,
346    ) -> Option<embedded_graphics_core::geometry::Point> {
347        let arr = [point.x, point.y, point.z];
348        let pt3 = self.transform_point(&arr, self.camera.vp_matrix)?;
349        Some(embedded_graphics_core::geometry::Point::new(pt3.x, pt3.y))
350    }
