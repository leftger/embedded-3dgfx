18#[cfg(feature = "lod-crossfade")]
19#[inline]
20pub(crate) fn apply_draw_alpha(prim: DrawPrimitive, alpha: u8) -> DrawPrimitive {
21    match prim {
22        DrawPrimitive::ColoredTriangleWithDepth {
23            points,
24            depths,
25            color,
26        } => DrawPrimitive::TranslucentTriangleWithDepth {
27            points,
28            depths,
29            color,
30            alpha,
31        },
32        DrawPrimitive::TranslucentTriangleWithDepth {
33            points,
34            depths,
35            color,
36            alpha: prev,
37        } => DrawPrimitive::TranslucentTriangleWithDepth {
38            points,
39            depths,
40            color,
41            alpha: ((prev as u16 * alpha as u16) / 255) as u8,
42        },
43        other => other,
44    }
45}

47#[cfg(feature = "lighting")]
48#[inline]
49pub(crate) fn light_tint_at(engine: &K3dengine, world_pos: Point3<f32>) -> Rgb565 {
50    let mut acc_r = 0u16;
51    let mut acc_g = 0u16;
52    let mut acc_b = 0u16;
53    for light in &engine.point_lights {
54        let tint = light.contribution_at(world_pos);
55        acc_r += tint.r() as u16;
56        acc_g += tint.g() as u16;
57        acc_b += tint.b() as u16;
58    }
59    Rgb565::new(
60        (acc_r.min(31)) as u8,
61        (acc_g.min(63)) as u8,
62        (acc_b.min(31)) as u8,
63    )
64}

66#[cfg(feature = "lighting")]
67#[inline]
68pub(crate) fn add_tint(base: Rgb565, tint: Rgb565) -> Rgb565 {
69    Rgb565::new(
70        (base.r() as u16 + tint.r() as u16).min(31) as u8,
71        (base.g() as u16 + tint.g() as u16).min(63) as u8,
72        (base.b() as u16 + tint.b() as u16).min(31) as u8,
73    )
74}

82#[cfg(feature = "lighting")]
83#[inline]
84pub(crate) fn sector_shaded_color(
85    engine: &K3dengine,
86    base: Rgb565,
87    brightness: u8,
88    face_center: Point3<f32>,
89) -> Rgb565 {
90    let level_u8 = match engine.light_levels {
91        crate::pipeline::shade::retro::light_levels::LightLevels::Linear => brightness,
92        crate::pipeline::shade::retro::light_levels::LightLevels::Doom32 => {
93            let base_level = (brightness as usize * 31) / 255;
94            let distance = (face_center - engine.camera.position).norm();
95            let distance_drop = (distance * 2.0) as usize;
96            let idx = base_level.saturating_sub(distance_drop).min(31);
97            DOOM_LIGHT_TABLE[idx]
98        }
99    };
100
101    let factor = level_u8 as f32 / 255.0;
102    Rgb565::new(
103        (base.r() as f32 * factor) as u8,
104        (base.g() as f32 * factor) as u8,
105        (base.b() as f32 * factor) as u8,
106    )
107}
