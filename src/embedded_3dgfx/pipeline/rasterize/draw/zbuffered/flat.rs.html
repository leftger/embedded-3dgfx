10#[inline]
11pub(crate) fn fill_triangle_zbuffered<D: DrawTarget<Color = Rgb565>>(
12    p1: nalgebra::Point2<i32>,
13    p2: nalgebra::Point2<i32>,
14    p3: nalgebra::Point2<i32>,
15    z1: f32,
16    z2: f32,
17    z3: f32,
18    color: Rgb565,
19    fb: &mut D,
20    zbuffer: &mut [crate::ZDepth],
21    width: usize,
22    fog_config: Option<&FogConfig>,
23    dither_config: Option<&DitherConfig>,
24) where
25    <D as DrawTarget>::Error: Debug,
26{
27    let p1_eg = Point::new(p1.x, p1.y);
28    let p2_eg = Point::new(p2.x, p2.y);
29    let p3_eg = Point::new(p3.x, p3.y);
30
31    let z1_int = (z1 * 65536.0) as u32;
32    let z2_int = (z2 * 65536.0) as u32;
33    let z3_int = (z3 * 65536.0) as u32;
34
35    if p2_eg.y == p3_eg.y {
36        fill_bottom_flat_triangle_zbuffered(
37            p1_eg,
38            p2_eg,
39            p3_eg,
40            z1_int,
41            z2_int,
42            z3_int,
43            color,
44            fb,
45            zbuffer,
46            width,
47            fog_config,
48            dither_config,
49        );
50    } else if p1_eg.y == p2_eg.y {
51        fill_top_flat_triangle_zbuffered(
52            p1_eg,
53            p2_eg,
54            p3_eg,
55            z1_int,
56            z2_int,
57            z3_int,
58            color,
59            fb,
60            zbuffer,
61            width,
62            fog_config,
63            dither_config,
64        );
65    } else {
66        let t = (p2_eg.y - p1_eg.y) as f32 / (p3_eg.y - p1_eg.y) as f32;
67        let p4 = Point::new(
68            (p1_eg.x as f32 + t * (p3_eg.x - p1_eg.x) as f32) as i32,
69            p2_eg.y,
70        );
71        let z4_int = (z1_int as i64 + (t * (z3_int as i64 - z1_int as i64) as f32) as i64) as u32;
72
73        fill_bottom_flat_triangle_zbuffered(
74            p1_eg,
75            p2_eg,
76            p4,
77            z1_int,
78            z2_int,
79            z4_int,
80            color,
81            fb,
82            zbuffer,
83            width,
84            fog_config,
85            dither_config,
86        );
87        fill_top_flat_triangle_zbuffered(
88            p2_eg,
89            p4,
90            p3_eg,
91            z2_int,
92            z4_int,
93            z3_int,
94            color,
95            fb,
96            zbuffer,
97            width,
98            fog_config,
99            dither_config,
100        );
101    }
102}

104#[inline]
105pub(crate) fn fill_bottom_flat_triangle_zbuffered<D: DrawTarget<Color = Rgb565>>(
106    p1: Point,
107    p2: Point,
108    p3: Point,
109    z1: u32,
110    z2: u32,
111    z3: u32,
112    color: Rgb565,
113    fb: &mut D,
114    zbuffer: &mut [crate::ZDepth],
115    width: usize,
116    fog_config: Option<&FogConfig>,
117    dither_config: Option<&DitherConfig>,
118) where
119    <D as DrawTarget>::Error: Debug,
120{
121    let height = p2.y - p1.y;
122    if height == 0 {
123        return;
124    }
125
126    let invslope1 = ((p2.x - p1.x) << 16) / height;
127    let invslope2 = ((p3.x - p1.x) << 16) / height;
128
129    let mut curx1 = (p1.x << 16) + (1 << 15);
130    let mut curx2 = (p1.x << 16) + (1 << 15);
131
132    let scr_h = (zbuffer.len() / width) as i32;
133    let y_skip = (0_i32 - p1.y).max(0);
134    curx1 = curx1.wrapping_add(invslope1.wrapping_mul(y_skip));
135    curx2 = curx2.wrapping_add(invslope2.wrapping_mul(y_skip));
136    let y_start = p1.y.max(0);
137    let y_end = p2.y.min(scr_h - 1);
138
139    for scanline_y in y_start..=y_end {
140        let dy = scanline_y - p1.y;
141        let z_left = if height > 0 {
142            (z1 as i64 + ((z2 as i64 - z1 as i64) * dy as i64 / height as i64)) as u32
143        } else {
144            z1
145        };
146        let z_right = if height > 0 {
147            (z1 as i64 + ((z3 as i64 - z1 as i64) * dy as i64 / height as i64)) as u32
148        } else {
149            z1
150        };
151
152        draw_scanline_zbuffered(
153            curx1 >> 16,
154            curx2 >> 16,
155            scanline_y,
156            z_left,
157            z_right,
158            color,
159            fb,
160            zbuffer,
161            width,
162            fog_config,
163            dither_config,
164        );
165
166        curx1 = curx1.wrapping_add(invslope1);
167        curx2 = curx2.wrapping_add(invslope2);
168    }
169}

171#[inline]
172pub(crate) fn fill_top_flat_triangle_zbuffered<D: DrawTarget<Color = Rgb565>>(
173    p1: Point,
174    p2: Point,
175    p3: Point,
176    z1: u32,
177    z2: u32,
178    z3: u32,
179    color: Rgb565,
180    fb: &mut D,
181    zbuffer: &mut [crate::ZDepth],
182    width: usize,
183    fog_config: Option<&FogConfig>,
184    dither_config: Option<&DitherConfig>,
185) where
186    <D as DrawTarget>::Error: Debug,
187{
188    let height = p3.y - p1.y;
189    if height == 0 {
190        return;
191    }
192
193    let invslope1 = ((p3.x - p1.x) << 16) / height;
194    let invslope2 = ((p3.x - p2.x) << 16) / height;
195
196    let mut curx1 = (p3.x << 16) + (1 << 15);
197    let mut curx2 = (p3.x << 16) + (1 << 15);
198
199    let scr_h = (zbuffer.len() / width) as i32;
200    let y_skip_bot = (p3.y - (scr_h - 1)).max(0);
201    curx1 = curx1.wrapping_sub(invslope1.wrapping_mul(y_skip_bot));
202    curx2 = curx2.wrapping_sub(invslope2.wrapping_mul(y_skip_bot));
203    let y_start = p1.y.max(0);
204    let y_end = p3.y.min(scr_h - 1);
205
206    for scanline_y in (y_start..=y_end).rev() {
207        let dy = scanline_y - p1.y;
208        let z_left = if height > 0 {
209            (z1 as i64 + ((z3 as i64 - z1 as i64) * dy as i64 / height as i64)) as u32
210        } else {
211            z1
212        };
213        let z_right = if height > 0 {
214            (z2 as i64 + ((z3 as i64 - z2 as i64) * dy as i64 / height as i64)) as u32
215        } else {
216            z2
217        };
218
219        draw_scanline_zbuffered(
220            curx1 >> 16,
221            curx2 >> 16,
222            scanline_y,
223            z_left,
224            z_right,
225            color,
226            fb,
227            zbuffer,
228            width,
229            fog_config,
230            dither_config,
231        );
232
233        curx1 = curx1.wrapping_sub(invslope1);
234        curx2 = curx2.wrapping_sub(invslope2);
235    }
236}

238#[inline(always)]
239pub(crate) fn draw_scanline_zbuffered<D: DrawTarget<Color = Rgb565>>(
240    x1: i32,
241    x2: i32,
242    y: i32,
243    z1: u32,
244    z2: u32,
245    color: Rgb565,
246    fb: &mut D,
247    zbuffer: &mut [crate::ZDepth],
248    width: usize,
249    fog_config: Option<&FogConfig>,
250    dither_config: Option<&DitherConfig>,
251) where
252    <D as DrawTarget>::Error: Debug,
253{
254    if y < 0 {
255        return;
256    }
257    let height = zbuffer.len() / width;
258    if y as usize >= height {
259        return;
260    }
261
262    let (left_x, right_x, z_left, z_right) = if x1 <= x2 {
263        (x1, x2, z1, z2)
264    } else {
265        (x2, x1, z2, z1)
266    };
267
268    let start_x = left_x.max(0);
269    let end_x = right_x.min(width as i32 - 1);
270    if start_x > end_x {
271        return;
272    }
273
274    let span = right_x - left_x;
275    let z_step = if span > 0 {
276        (((z_right as i64 - z_left as i64) << 16) / span as i64) as i32
277    } else {
278        0
279    };
280
281    let mut zbuf_idx = y as usize * width + start_x as usize;
282
283    if z_step == 0 {
284        let z = z_left;
285        let z_depth = crate::to_zdepth(z);
286        let mut base_color = color;
287        if let Some(fog) = fog_config {
288            base_color = fog.apply(base_color, z);
289        }
290        for x in start_x..=end_x {
291            if z_depth < zbuffer[zbuf_idx].saturating_add(crate::DEPTH_EPSILON) {
292                zbuffer[zbuf_idx] = z_depth;
293                let final_color = if let Some(dither) = dither_config {
294                    dither.apply(base_color, x, y)
295                } else {
296                    base_color
297                };
298                fb.draw_iter([embedded_graphics_core::Pixel(Point::new(x, y), final_color)])
299                    .unwrap();
300            }
301            zbuf_idx += 1;
302        }
303        return;
304    }
305
306    let left_clip = start_x - left_x;
307    let mut z_curr = ((z_left as i64) << 16) + (left_clip as i64 * z_step as i64);
308
309    for x in start_x..=end_x {
310        let z = (z_curr >> 16) as u32;
311        z_curr += z_step as i64;
312        let z_depth = crate::to_zdepth(z);
313
314        if z_depth < zbuffer[zbuf_idx].saturating_add(crate::DEPTH_EPSILON) {
315            zbuffer[zbuf_idx] = z_depth;
316
317            let mut final_color = color;
318
319            if let Some(fog) = fog_config {
320                final_color = fog.apply(final_color, z);
321            }
322
323            if let Some(dither) = dither_config {
324                final_color = dither.apply(final_color, x, y);
325            }
326
327            fb.draw_iter([embedded_graphics_core::Pixel(Point::new(x, y), final_color)])
328                .unwrap();
329        }
330        zbuf_idx += 1;
331    }
