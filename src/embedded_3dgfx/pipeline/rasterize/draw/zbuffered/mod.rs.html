23/// Render a primitive with Z-buffering and default raster state.
24#[inline]
25pub fn draw_zbuffered<D: DrawTarget<Color = Rgb565>>(
26    primitive: DrawPrimitive,
27    fb: &mut D,
28    zbuffer: &mut [crate::ZDepth],
29    width: usize,
30) where
31    <D as DrawTarget>::Error: Debug,
32{
33    let state = RasterState::from_zbuffer(width, zbuffer.len());
34    draw_zbuffered_with_state(primitive, fb, zbuffer, &state);
35}

37/// Apply the per-triangle depth interpolation mode and optional constant bias.
38#[inline]
39fn resolve_depths(
40    raw: [f32; 3],
41    points: &[nalgebra::Point2<i32>; 3],
42    state: &RasterState<'_>,
43) -> (f32, f32, f32) {
44    let (z1, z2, z3) = state.depth_mode.process_depths(raw[0], raw[1], raw[2]);
45    match state.depth_bias {
46        Some(b) => b.apply(points[0], points[1], points[2], z1, z2, z3),
47        None => (z1, z2, z3),
48    }
49}

51/// Render a primitive with Z-buffering, using fog / dither / depth settings
52/// from `state`.
53///
54/// This is the single entry point the old `draw_zbuffered`,
55/// `_with_effects`, `_with_options`, and `_with_bias` wrappers all forwarded to;
56/// they are now expressed as [`RasterState`] builders instead of a chain of
57/// near-identical functions.
58pub fn draw_zbuffered_with_state<D: DrawTarget<Color = Rgb565>>(
59    primitive: DrawPrimitive,
60    fb: &mut D,
61    zbuffer: &mut [crate::ZDepth],
62    state: &RasterState<'_>,
63) where
64    <D as DrawTarget>::Error: Debug,
65{
66    match primitive {
67        DrawPrimitive::ColoredTriangleWithDepth {
68            mut points,
69            depths: mut raw_depths,
70            color,
71        } => {
72            let order = ascending_y_order(&points);
73            apply_order(&mut points, order);
74            apply_order(&mut raw_depths, order);
75            if tri_fully_off_screen(&points, state.width, state.height) {
76                return;
77            }
78            let (z1, z2, z3) = resolve_depths(raw_depths, &points, state);
79            let [p1, p2, p3] = points;
80
81            // Hardware path: offer the triangle to the sink before rasterizing.
82            // The sink sees the same vertices and the same *resolved* depths the
83            // CPU path would use, so a backend that declines -- buffer full, or a
84            // primitive it cannot represent -- gets an identical CPU fallback.
85            //
86            // Withheld when fog or dither is active, because the CPU path applies
87            // both *inside* `fill_triangle_zbuffered`, below. Offering the triangle
88            // anyway would render it without them: wrong output rather than a slow
89            // one. Screen-door stipple is a separate primitive variant, so it is not
90            // a concern here. A sink able to apply these effects itself cannot yet
91            // say so -- that needs a capability handshake rather than an assumption.
92            let effects_blocked = state.fog.is_some() || state.dither.is_some();
93            if let Some(sink) = state.sink {
94                if !effects_blocked && sink.triangle(&[p1, p2, p3], &[z1, z2, z3], color) {
95                    return;
96                }
97            }
98
99            fill_triangle_zbuffered(
100                p1,
101                p2,
102                p3,
103                z1,
104                z2,
105                z3,
106                color,
107                fb,
108                zbuffer,
109                state.width,
110                state.fog,
111                state.dither,
112            );
113        }
114
115        DrawPrimitive::TranslucentTriangleWithDepth {
116            mut points,
117            depths: mut raw_depths,
118            color,
119            alpha,
120        } => {
121            let order = ascending_y_order(&points);
122            apply_order(&mut points, order);
123            apply_order(&mut raw_depths, order);
124            if tri_fully_off_screen(&points, state.width, state.height) {
125                return;
126            }
127            let (z1, z2, z3) = resolve_depths(raw_depths, &points, state);
128            let [p1, p2, p3] = points;
129            fill_triangle_zbuffered_translucent(
130                p1,
131                p2,
132                p3,
133                z1,
134                z2,
135                z3,
136                color,
137                alpha,
138                fb,
139                zbuffer,
140                state.width,
141            );
142        }
143
144        DrawPrimitive::ScreenDoorTriangleWithDepth {
145            mut points,
146            depths: mut raw_depths,
147            color,
148            alpha,
149        } => {
150            let order = ascending_y_order(&points);
151            apply_order(&mut points, order);
152            apply_order(&mut raw_depths, order);
153            if tri_fully_off_screen(&points, state.width, state.height) {
154                return;
155            }
156            let (z1, z2, z3) = resolve_depths(raw_depths, &points, state);
157            let [p1, p2, p3] = points;
158            fill_triangle_zbuffered_screendoor(
159                p1,
160                p2,
161                p3,
162                z1,
163                z2,
164                z3,
165                color,
166                alpha,
167                fb,
168                zbuffer,
169                state.width,
170            );
171        }
172
173        #[cfg(feature = "lighting")]
174        DrawPrimitive::GouraudTriangleWithDepth {
175            mut points,
176            depths: mut raw_depths,
177            mut colors,
178        } => {
179            let order = ascending_y_order(&points);
180            apply_order(&mut points, order);
181            apply_order(&mut raw_depths, order);
182            apply_order(&mut colors, order);
183            if tri_fully_off_screen(&points, state.width, state.height) {
184                return;
185            }
186            let (z1, z2, z3) = resolve_depths(raw_depths, &points, state);
187            let [p1, p2, p3] = points;
188            let [c1, c2, c3] = colors;
189            fill_triangle_zbuffered_gouraud(
190                p1,
191                p2,
192                p3,
193                z1,
194                z2,
195                z3,
196                c1,
197                c2,
198                c3,
199                fb,
200                zbuffer,
201                state.width,
202                state.fog,
203                state.dither,
204            );
205        }
206
207        // Non-zbuffered primitives -- lines, points, plain triangles -- are offered
208        // to the sink inside `fill::draw_with_state`, which is why the state has to
209        // travel with them.
210        _ => super::fill::draw_with_state(primitive, fb, state),
211    }
212}

214/// Compatibility adapter over [`draw_zbuffered_with_state`]: Z-buffer render
215/// with optional fog / dither post-processing.
216#[inline]
217pub fn draw_zbuffered_with_effects<D: DrawTarget<Color = Rgb565>>(
218    primitive: DrawPrimitive,
219    fb: &mut D,
220    zbuffer: &mut [crate::ZDepth],
221    width: usize,
222    fog_config: Option<&FogConfig>,
223    dither_config: Option<&DitherConfig>,
224) where
225    <D as DrawTarget>::Error: Debug,
226{
227    let state = RasterState::from_zbuffer(width, zbuffer.len())
228        .with_fog(fog_config)
229        .with_dither(dither_config);
230    draw_zbuffered_with_state(primitive, fb, zbuffer, &state);
231}

304    impl<const W: usize, const H: usize> DrawTarget for TestFb<W, H> {
305        type Color = Rgb565;
306        type Error = core::convert::Infallible;
307
308        fn draw_iter<I>(&mut self, pixels: I) -> Result<(), Self::Error>
309        where
310            I: IntoIterator<Item = Pixel<Self::Color>>,
311        {
312            for Pixel(point, color) in pixels {
313                if point.x >= 0 && point.x < W as i32 && point.y >= 0 && point.y < H as i32 {
314                    let idx = (point.y as usize) * W + (point.x as usize);
315                    if idx < self.pixels.len() {
316                        self.pixels[idx] = color;
317                    }
318                }
319            }
320            Ok(())
321        }
322    }

360    /// Records what it is offered, and can be told to decline.
361    #[derive(Debug)]
362    struct RecordingSink {
363        accept: bool,
364        seen: core::cell::Cell<usize>,
365        last: core::cell::Cell<(nalgebra::Point2<i32>, f32, Rgb565)>,
366    }
367
368    impl RecordingSink {
369        fn new(accept: bool) -> Self {
370            Self {
371                accept,
372                seen: core::cell::Cell::new(0),
373                last: core::cell::Cell::new((nalgebra::Point2::new(0, 0), 0.0, Rgb565::BLACK)),
374            }
375        }
376    }
377
378    impl crate::pipeline::rasterize::draw::sink::RasterSink for RecordingSink {
379        fn triangle(
380            &self,
381            points: &[nalgebra::Point2<i32>; 3],
382            depths: &[f32; 3],
383            color: Rgb565,
384        ) -> bool {
385            self.seen.set(self.seen.get() + 1);
386            self.last.set((points[0], depths[0], color));
387            self.accept
388        }
389
390        fn line(&self, a: nalgebra::Point2<i32>, _b: nalgebra::Point2<i32>, color: Rgb565) -> bool {
391            self.seen.set(self.seen.get() + 1);
392            self.last.set((a, 0.0, color));
393            self.accept
394        }
395    }

397    /// The sink sees the triangle the CPU would have drawn, and its answer
398    /// decides whether the CPU draws it.
399    #[test]
400    fn raster_sink_takes_over_and_can_decline() {
401        let tri = || DrawPrimitive::ColoredTriangleWithDepth {
402            points: [
403                nalgebra::Point2::new(10, 2),
404                nalgebra::Point2::new(2, 18),
405                nalgebra::Point2::new(18, 18),
406            ],
407            depths: [1.0, 1.0, 1.0],
408            color: Rgb565::RED,
409        };
410        let center = 10 * 20 + 10;
411
412        // Accepted: hardware owns the triangle, so the CPU must not draw it.
413        let mut fb = TestFb::<20, 20>::default();
414        let mut zbuf = [crate::Z_MAX_VALUE; 400];
415        let accepting = RecordingSink::new(true);
416        let state = RasterState::new(20, 20).with_raster_sink(Some(&accepting));
417        draw_zbuffered_with_state(tri(), &mut fb, &mut zbuf, &state);
418
419        assert_eq!(
420            accepting.seen.get(),
421            1,
422            "triangle should be offered to the sink"
423        );
424        let (first_point, first_depth, color) = accepting.last.get();
425        assert_eq!(first_point, nalgebra::Point2::new(10, 2));
426        assert_eq!(first_depth, 1.0);
427        assert_eq!(color, Rgb565::RED);
428        assert_eq!(
429            fb.pixels[center],
430            Rgb565::BLACK,
431            "an accepted triangle must be left to the hardware"
432        );
433
434        // Declined: the CPU rasterizer must still draw it.
435        let mut fb2 = TestFb::<20, 20>::default();
436        let mut zbuf2 = [crate::Z_MAX_VALUE; 400];
437        let declining = RecordingSink::new(false);
438        let state2 = RasterState::new(20, 20).with_raster_sink(Some(&declining));
439        draw_zbuffered_with_state(tri(), &mut fb2, &mut zbuf2, &state2);
440
441        assert_eq!(
442            declining.seen.get(),
443            1,
444            "a declining sink is still offered it"
445        );
446        assert_eq!(
447            fb2.pixels[center],
448            Rgb565::RED,
449            "a declined triangle must fall back to the CPU rasterizer"
450        );
451    }
