12/// Fast RGB565 alpha blending using 32-bit SWAR (SIMD within a register).
13///
14/// Blends Red and Blue channels in parallel in one 32-bit register and Green in another,
15/// eliminating per-channel bit unpack overhead and integer divisions.
16#[inline(always)]
17pub fn fast_blend_rgb565(bg: Rgb565, fg: Rgb565, alpha: u8) -> Rgb565 {
18    if alpha == 255 {
19        return fg;
20    }
21    if alpha == 0 {
22        return bg;
23    }
24
25    let bg_raw = bg.into_storage() as u32;
26    let fg_raw = fg.into_storage() as u32;
27    let a = alpha as u32;
28
29    // Mask R & B: 0b11111_000000_11111 = 0xF81F
30    // Mask G:   0b00000_111111_00000 = 0x07E0
31    let bg_rb = bg_raw & 0xF81F;
32    let fg_rb = fg_raw & 0xF81F;
33    let bg_g = bg_raw & 0x07E0;
34    let fg_g = fg_raw & 0x07E0;
35
36    // Linear interpolation with 1/256 rounding:
37    // val = bg + ((fg - bg) * a + 128) >> 8
38    let rb = ((fg_rb.wrapping_sub(bg_rb))
39        .wrapping_mul(a)
40        .wrapping_add(0x8010)
41        >> 8)
42        .wrapping_add(bg_rb)
43        & 0xF81F;
44    let g = ((fg_g.wrapping_sub(bg_g))
45        .wrapping_mul(a)
46        .wrapping_add(0x0400)
47        >> 8)
48        .wrapping_add(bg_g)
49        & 0x07E0;
50
51    Rgb565::from(RawU16::new((rb | g) as u16))
52}

54/// Fast RGBA8888 alpha blending with division-free 8-bit scaling.
55#[inline(always)]
56pub fn fast_blend_rgba8888(bg: [u8; 4], fg: [u8; 4]) -> [u8; 4] {
57    let a = fg[3] as u32;
58    if a == 255 {
59        return fg;
60    }
61    if a == 0 {
62        return bg;
63    }
64    let inv = 255 - a;
65    // Division-free scaling by 255: (x + 1 + (x >> 8)) >> 8
66    let blend_ch = |b: u8, f: u8| -> u8 {
67        let prod = (b as u32) * inv + (f as u32) * a;
68        ((prod + 1 + (prod >> 8)) >> 8) as u8
69    };
70    let r = blend_ch(bg[0], fg[0]);
71    let g = blend_ch(bg[1], fg[1]);
72    let b = blend_ch(bg[2], fg[2]);
73    let out_a_prod = (bg[3] as u32) * inv;
74    let out_a = fg[3] as u32 + ((out_a_prod + 1 + (out_a_prod >> 8)) >> 8);
75    [r, g, b, out_a as u8]
76}

78/// Fast RGBA8888 to RGB565 alpha blending.
79#[inline(always)]
80pub fn fast_blend_rgba8888_to_rgb565(bg: Rgb565, fg_rgba: [u8; 4]) -> Rgb565 {
81    let alpha = fg_rgba[3];
82    if alpha == 0 {
83        return bg;
84    }
85    let fg_r = fg_rgba[0] >> 3;
86    let fg_g = fg_rgba[1] >> 2;
87    let fg_b = fg_rgba[2] >> 3;
88    let fg_565 = Rgb565::new(fg_r, fg_g, fg_b);
89    fast_blend_rgb565(bg, fg_565, alpha)
90}

92/// Fast color inversion (reverse color) filter for RGB565.
93#[inline(always)]
94pub fn reverse_color_rgb565(c: Rgb565) -> Rgb565 {
95    Rgb565::new(31 - c.r(), 63 - c.g(), 31 - c.b())
96}

98/// Fast color inversion (reverse color) filter for RGBA8888.
99#[inline(always)]
100pub fn reverse_color_rgba8888(rgba: [u8; 4]) -> [u8; 4] {
101    [255 - rgba[0], 255 - rgba[1], 255 - rgba[2], rgba[3]]
102}

104/// Component-wise blend in 8-bit fixed-point coverage.
105#[cfg(feature = "aa")]
106#[inline(always)]
107pub(crate) fn blend_q8(bg: Rgb565, fg: Rgb565, coverage_q8: u32) -> Rgb565 {
108    let inv = 256 - coverage_q8;
109    let r = (bg.r() as u32 * inv + fg.r() as u32 * coverage_q8) >> 8;
110    let g = (bg.g() as u32 * inv + fg.g() as u32 * coverage_q8) >> 8;
111    let b = (bg.b() as u32 * inv + fg.b() as u32 * coverage_q8) >> 8;
112    Rgb565::new(r as u8, g as u8, b as u8)
113}
