1//! Pipeline stage 5 — output.
2//!
3//! Framebuffer backends ([`display_backend`]), triple-buffered presentation
4//! ([`swapchain`]), async DMA completion slots ([`completion`]), and the 2D
5//! [`hud`] overlay composited over the rendered frame.
6//!
7//! # Writing a display backend
8//!
9//! The traits below name types from `embedded-graphics-framebuf` in their
10//! signatures, so they are re-exported here: a driver needs exactly one `use`:
11//!
12//! ```
13//! use embedded_3dgfx::pipeline::output::{
14//!     DisplayBackend, DisplayRegion, DmaTransfer, FrameBuf, StandardSwapChain, TransferError,
15//! };
16//! ```
17
18pub mod completion;
19pub mod display_backend;
20#[cfg(feature = "hud")]
21pub mod hud;
22pub mod swapchain;
23
24// Stage facade: the vocabulary for driving a panel, gathered so that a board
25// driver imports one path instead of reaching into four submodules. Individual
26// submodules stay minimal — this is the deliberate, curated surface for the
27// stage as a whole.
28pub use completion::{CompletionSlot, WaitTransfer};
29pub use display_backend::{
30    AsyncDmaTransfer, DisplayBackend, DisplayRegion, DmaTransfer, HardwareAccelerator,
31    SimulatorBackend, TransferError,
32};
33pub use swapchain::{StandardSwapChain, SwapChain};
34
35// Re-exported because `DisplayBackend` and `DmaTransfer` name them in their
36// signatures: an implementor must be able to spell these types without adding
37// `embedded-graphics-framebuf` to its own dependencies.
38pub use embedded_graphics_framebuf::{FrameBuf, backends::DMACapableFrameBufferBackend};
39
40#[cfg(feature = "triple-buffering")]
41pub use swapchain::{StandardTripleSwapChain, TripleSwapChain};
