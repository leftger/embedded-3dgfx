18#[inline]
19pub(super) fn stripe_on_at(x: i32, scroll: i32, stripe_w: i32) -> bool {
20    (((x + scroll).div_euclid(stripe_w)) & 1) == 0
21}
22
23pub(super) fn draw_sky_background<D>(
24    fb: &mut D,
25    width: usize,
26    height: usize,
27    sky: SkyConfig,
28    camera_dir: [f32; 3],
29    screen_tint: Option<ScreenTint>,
30    palette_mode: PaletteMode,
31) -> Result<(), RenderError>
32where
33    D: DrawTarget<Color = Rgb565> + OriginDimensions,
34    D::Error: Debug,
35{
36    let w = width as i32;
37    let h = height as i32;
38    if w <= 0 || h <= 0 {
39        return Ok(());
40    }
41
42    let horizon = (h as f32 * (0.5 + camera_dir[1].clamp(-1.0, 1.0) * 0.25)) as i32;
43    let stripe_w = sky.stripe_width.max(1) as i32;
44    let scroll = (camera_dir[0] * 128.0) as i32;
45    let stripe_fade_span = (h / 6).max(1);
46
47    for y in 0..h {
48        let dy = (y - horizon + h / 2).clamp(0, h);
49        let t_q8 = ((dy as i64 * 255) / h.max(1) as i64) as u16;
50        let base = blend_rgb565(sky.top_color, sky.bottom_color, t_q8);
51        let below_horizon = (y - horizon).max(0);
52        let stripe_strength = if below_horizon == 0 {
53            sky.stripe_strength as u16
54        } else if below_horizon >= stripe_fade_span {
55            0
56        } else {
57            let rem = stripe_fade_span - below_horizon;
58            ((sky.stripe_strength as i32 * rem) / stripe_fade_span) as u16
59        };
60        for x in 0..w {
61            let stripe_on = stripe_on_at(x, scroll, stripe_w);
62            let mut color = if stripe_on && stripe_strength > 0 {
63                blend_rgb565(base, sky.stripe_color, stripe_strength)
64            } else {
65                base
66            };
67            color = apply_post(color, screen_tint, palette_mode);
68            fb.draw_iter([Pixel(Point::new(x, y), color)])
69                .map_err(|_| RenderError::InvalidInput("draw target rejected sky write"))?;
70        }
71    }
72    Ok(())
73}
